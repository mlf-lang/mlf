  isClosed :: XType -> Bool;
  isClosed ty = closedIn S.empty ty;
  closedIn :: Set XTyVar -> XType -> Bool;
  closedIn scope ty = case ty of {
    XVar a -> S.member compareTyVar a scope;
    XBot -> True;
    XBotKind _ -> True;
    XArrow a b -> closedIn scope a && closedIn scope b;
    XCon _ arguments -> L.all (closedIn scope) arguments;
    XApply f a -> closedIn scope f && closedIn scope a;
    XForall a bound body ->
      closedIn scope bound && closedIn (S.insert compareTyVar a scope) body
  };
