{-# LANGUAGE NoImplicitPrelude, NoMonomorphismRestriction, RankNTypes,
             KindSignatures, FlexibleContexts, FlexibleInstances,
             MultiParamTypeClasses, OverloadedStrings #-}
module Main where {
  import MLF.Primitive;
  import Bootstrap.Types;
  import Bootstrap.Std;
  -- INSERT_ACTUAL_PACKED_WRITERS
  dump bytes index = if index == 16 then ioReturn () else do {
    readByteArray bytes index >>= printInt;
    dump bytes (index + 1)
  };
  main = do {
    mode <- getArg 0; usedText <- getArg 1; pendingText <- getArg 2;
    widthText <- getArg 3; bitsText <- getArg 4;
    let { used = readInt usedText; pending = readInt pendingText;
      width = readInt widthText; bits = readInt bitsText };
    bytes <- newByteArray 16 211;
    let { finish i u p = do {
      printInt i; printInt u; printInt p; packedFinish bytes i u p
    } };
    end <- if eqString mode "byte" then packedByte bytes 1 used pending bits finish
      else if eqString mode "var" then packedVar bytes 1 used pending bits finish
      else packedBits bytes 1 used pending width bits finish;
    printInt end;
    dump bytes 0
  }
}
